Understand the Table Relationships
Before creating tables, understand how they connect. This is called an Entity Relationship — every piece of data links to another:
DATA RELATIONSHIPS
👤 Users (Admins)
↓ owns
🏗️ Plots
↓ contains
🏠 Houses / Units
↓ occupied by
👥 Tenants
💳 Payments
🔧 Maintenance
3
Create All Database Tables
In your backend folder, create a file called database/schema.sql and paste this SQL. Then run it in PostgreSQL:
database/schema.sql
-- 1. USERS TABLE (Admins, Managers, Tenants) CREATE TABLE users ( id SERIAL PRIMARY KEY, name VARCHAR(100) NOT NULL, email VARCHAR(100) UNIQUE NOT NULL, password VARCHAR(255) NOT NULL, -- stored hashed, never plain text role VARCHAR(20) DEFAULT 'admin', -- admin | manager | tenant phone VARCHAR(20), created_at TIMESTAMP DEFAULT NOW() ); -- 2. PLOTS TABLE (Properties / Buildings) CREATE TABLE plots ( id SERIAL PRIMARY KEY, user_id INTEGER REFERENCES users(id), name VARCHAR(100) NOT NULL, location VARCHAR(200), description TEXT, gps_lat DECIMAL(10,8), gps_lng DECIMAL(11,8), created_at TIMESTAMP DEFAULT NOW() ); -- 3. HOUSES TABLE (Individual rental units) CREATE TABLE houses ( id SERIAL PRIMARY KEY, plot_id INTEGER REFERENCES plots(id) ON DELETE CASCADE, unit_number VARCHAR(20) NOT NULL, house_type VARCHAR(50), -- Bedsitter, 1 Bed, 2 Bed, Shop rent_amount DECIMAL(10,2) NOT NULL, deposit DECIMAL(10,2), status VARCHAR(20) DEFAULT 'vacant', -- vacant | occupied meter_no VARCHAR(50), created_at TIMESTAMP DEFAULT NOW() ); -- 4. TENANTS TABLE CREATE TABLE tenants ( id SERIAL PRIMARY KEY, user_id INTEGER REFERENCES users(id), -- tenant's login account house_id INTEGER REFERENCES houses(id), full_name VARCHAR(100) NOT NULL, national_id VARCHAR(30), phone VARCHAR(20), email VARCHAR(100), emergency_phone VARCHAR(20), move_in_date DATE, rent_due_day INTEGER DEFAULT 5, deposit_paid DECIMAL(10,2) DEFAULT 0, balance DECIMAL(10,2) DEFAULT 0, -- outstanding rent photo_url VARCHAR(255), lease_url VARCHAR(255), is_active BOOLEAN DEFAULT true, created_at TIMESTAMP DEFAULT NOW() ); -- 5. PAYMENTS TABLE (All M-Pesa transactions) CREATE TABLE payments ( id SERIAL PRIMARY KEY, tenant_id INTEGER REFERENCES tenants(id), transaction_id VARCHAR(50) UNIQUE, -- M-Pesa transaction code amount DECIMAL(10,2) NOT NULL, phone VARCHAR(20), payment_date TIMESTAMP DEFAULT NOW(), house_id INTEGER REFERENCES houses(id), method VARCHAR(30) DEFAULT 'mpesa', status VARCHAR(20) DEFAULT 'unmatched', -- matched | unmatched | review match_score INTEGER DEFAULT 0, -- 0-100 confidence receipt_no VARCHAR(50), notes TEXT ); -- 6. EXPENSES TABLE CREATE TABLE expenses ( id SERIAL PRIMARY KEY, plot_id INTEGER REFERENCES plots(id), category VARCHAR(50), -- Repairs, Electricity, Water, Security etc. description TEXT, amount DECIMAL(10,2), expense_date DATE, receipt_url VARCHAR(255), created_at TIMESTAMP DEFAULT NOW() ); -- 7. MAINTENANCE REQUESTS CREATE TABLE maintenance_requests ( id SERIAL PRIMARY KEY, tenant_id INTEGER REFERENCES tenants(id), house_id INTEGER REFERENCES houses(id), issue TEXT NOT NULL, photo_url VARCHAR(255), priority VARCHAR(20) DEFAULT 'medium', status VARCHAR(20) DEFAULT 'pending', -- pending | in_progress | completed reported_at TIMESTAMP DEFAULT NOW(), resolved_at TIMESTAMP ); -- 8. SMS LOGS CREATE TABLE sms_logs ( id SERIAL PRIMARY KEY, to_phone VARCHAR(20), message TEXT, status VARCHAR(20), -- sent | failed sent_at TIMESTAMP DEFAULT NOW() ); -- Run this file in psql: -- \i database/schema.sql

⚙️ Backend API with Node.js

Build the server that powers all your data — properties, tenants, payments, and more.

1
Create the Main Server File
This is the heart of your backend. Create backend/server.js:
backend/server.js
const express = require('express'); const cors = require('cors'); require('dotenv').config(); const app = express(); // Middleware — process incoming requests app.use(cors()); // Allow cross-origin requests from React app.use(express.json()); // Parse JSON request bodies app.use('/uploads', express.static('uploads')); // Serve uploaded files // Import route files (we'll create these next) app.use('/api/auth', require('./routes/auth')); app.use('/api/plots', require('./routes/plots')); app.use('/api/houses', require('./routes/houses')); app.use('/api/tenants', require('./routes/tenants')); app.use('/api/payments', require('./routes/payments')); app.use('/api/expenses', require('./routes/expenses')); app.use('/api/maintenance', require('./routes/maintenance')); app.use('/api/mpesa', require('./routes/mpesa')); app.use('/api/sms', require('./routes/sms')); app.use('/api/listings', require('./routes/listings')); // public portal // Health check — visit http://localhost:5000/api/health app.get('/api/health', (req, res) => { res.json({ status: 'ok', message: 'RentFlow API running!' }); }); const PORT = process.env.PORT || 5000; app.listen(PORT, () => console.log(`🚀 Server running on port ${PORT}`));
2
Build User Authentication (Login & Register)
Create backend/routes/auth.js. This handles secure login for admins and tenants:
backend/routes/auth.js
const router = require('express').Router(); const db = require('../db'); const bcrypt = require('bcryptjs'); const jwt = require('jsonwebtoken'); // POST /api/auth/register — Create new admin account router.post('/register', async (req, res) => { const { name, email, password, role } = req.body; try { // Hash password (NEVER store plain text passwords!) const hashedPassword = await bcrypt.hash(password, 10); const result = await db.query( 'INSERT INTO users (name, email, password, role) VALUES ($1,$2,$3,$4) RETURNING id,name,email,role', [name, email, hashedPassword, role || 'admin'] ); res.json({ success: true, user: result.rows[0] }); } catch (err) { res.status(400).json({ error: err.message }); } }); // POST /api/auth/login — Login and get a token router.post('/login', async (req, res) => { const { email, password } = req.body; try { const result = await db.query('SELECT * FROM users WHERE email = $1', [email]); const user = result.rows[0]; if (!user) return res.status(401).json({ error: 'User not found' }); // Compare entered password with stored hash const valid = await bcrypt.compare(password, user.password); if (!valid) return res.status(401).json({ error: 'Wrong password' }); // Create a token (like a digital key card) const token = jwt.sign( { id: user.id, role: user.role }, process.env.JWT_SECRET, { expiresIn: '7d' } // expires in 7 days ); res.json({ token, user: { id: user.id, name: user.name, role: user.role } }); } catch (err) { res.status(500).json({ error: err.message }); } }); module.exports = router;
3
Build the Tenants Route
Create backend/routes/tenants.js. This pattern repeats for all routes:
backend/routes/tenants.js (example pattern)
const router = require('express').Router(); const db = require('../db'); const auth = require('../middleware/auth'); // protects routes // GET all tenants — GET /api/tenants router.get('/', auth, async (req, res) => { const result = await db.query(` SELECT t.*, h.unit_number, h.rent_amount, p.name as plot_name FROM tenants t LEFT JOIN houses h ON t.house_id = h.id LEFT JOIN plots p ON h.plot_id = p.id WHERE t.is_active = true ORDER BY t.full_name `); res.json(result.rows); }); -- GET single tenant with payment history router.get('/:id', auth, async (req, res) => { const tenant = await db.query('SELECT * FROM tenants WHERE id=$1', [req.params.id]); const payments = await db.query( 'SELECT * FROM payments WHERE tenant_id=$1 ORDER BY payment_date DESC', [req.params.id] ); res.json({ ...tenant.rows[0], payments: payments.rows }); }); // POST add new tenant — POST /api/tenants router.post('/', auth, async (req, res) => { const { full_name, phone, email, house_id, move_in_date, deposit_paid } = req.body; const result = await db.query( `INSERT INTO tenants (full_name, phone, email, house_id, move_in_date, deposit_paid) VALUES ($1,$2,$3,$4,$5,$6) RETURNING *`, [full_name, phone, email, house_id, move_in_date, deposit_paid] ); // Mark the house as occupied await db.query("UPDATE houses SET status='occupied' WHERE id=$1", [house_id]); res.json(result.rows[0]); }); module.exports = router;
Follow this exact same pattern to create plots.js, houses.js, expenses.js, and maintenance.js. The structure is always: GET all, GET one, POST create, PUT update, DELETE remove.

Sandbox = pretend payments for testing. Live = real money. Always test in Sandbox first until everything works perfectly!
2
Understand How M-Pesa STK Push Works
STK Push = Safaricom sends a payment prompt to the tenant's phone automatically. Here's the flow:
1
Tenant clicks "Pay Rent" button on the portal
2
Your backend sends a request to Safaricom's API with the phone number + amount
3
Safaricom sends a PIN prompt to the tenant's phone
4
Tenant enters M-Pesa PIN and confirms
5
Safaricom calls your callback URL — your system records the payment automatically ✅
3
Build the M-Pesa Route
Create backend/routes/mpesa.js:
backend/routes/mpesa.js
const router = require('express').Router(); const axios = require('axios'); const db = require('../db'); // Step 1: Get an access token from Safaricom async function getMpesaToken() { const key = process.env.MPESA_CONSUMER_KEY; const secret = process.env.MPESA_CONSUMER_SECRET; const auth = Buffer.from(`${key}:${secret}`).toString('base64'); const response = await axios.get( 'https://sandbox.safaricom.co.ke/oauth/v1/generate?grant_type=client_credentials', { headers: { Authorization: `Basic ${auth}` } } ); return response.data.access_token; } // Step 2: Initiate STK Push (send payment prompt to phone) router.post('/stk-push', async (req, res) => { const { phone, amount, accountRef } = req.body; const token = await getMpesaToken(); const timestamp = new Date().toISOString().replace(/[^0-9]/g, '').slice(0, 14); const password = Buffer.from( process.env.MPESA_PAYBILL + process.env.MPESA_PASSKEY + timestamp ).toString('base64'); const response = await axios.post( 'https://sandbox.safaricom.co.ke/mpesa/stkpush/v1/processrequest', { BusinessShortCode: process.env.MPESA_PAYBILL, Password: password, Timestamp: timestamp, TransactionType: 'CustomerPayBillOnline', Amount: amount, PartyA: phone, // tenant phone PartyB: process.env.MPESA_PAYBILL, PhoneNumber: phone, CallBackURL: process.env.MPESA_CALLBACK_URL, AccountReference: accountRef, // house unit number TransactionDesc: 'Rent Payment' }, { headers: { Authorization: `Bearer ${token}` } } ); res.json(response.data); }); // Step 3: Receive callback from Safaricom when payment succeeds router.post('/callback', async (req, res) => { const callback = req.body.Body.stkCallback; if (callback.ResultCode === 0) { // Payment succeeded! Extract details const items = callback.CallbackMetadata.Item; const amount = items.find(i => i.Name === 'Amount').Value; const txnId = items.find(i => i.Name === 'MpesaReceiptNumber').Value; const phone = items.find(i => i.Name === 'PhoneNumber').Value.toString(); // Save payment to database await db.query( 'INSERT INTO payments (transaction_id, amount, phone, status) VALUES ($1,$2,$3,$4)', [txnId, amount, phone, 'unmatched'] ); // Try to auto-match to a tenant await autoMatchPayment(txnId, phone, amount); } res.json({ ResultCode: 0, ResultDesc: 'Accepted' }); // Always send this back! }); // Auto-match payment to tenant using phone number async function autoMatchPayment(txnId, phone, amount) { const tenant = await db.query( 'SELECT * FROM tenants WHERE phone = $1 AND is_active = true', [phone.slice(-9)] // match last 9 digits ); if (tenant.rows.length > 0) { const t = tenant.rows[0]; // Update payment with tenant match await db.query( 'UPDATE payments SET tenant_id=$1, house_id=$2, status=$3, match_score=$4 WHERE transaction_id=$5', [t.id, t.house_id, 'matched', 95, txnId] ); // Update tenant balance await db.query( 'UPDATE tenants SET balance = GREATEST(0, balance - $1) WHERE id = $2', [amount, t.id] ); } } module.exports = router;

Build the SMS Service

backend/services/sms.js
const axios = require('axios'); const db = require('../db'); require('dotenv').config(); async function sendSMS(phone, message) { try { const response = await axios.post( 'https://api.africastalking.com/version1/messaging', new URLSearchParams({ username: process.env.AT_USERNAME, to: `+254${phone.slice(-9)}`, // format: +254712345678 message: message, from: process.env.AT_SENDER_ID }), { headers: { apiKey: process.env.AT_API_KEY } } ); // Save to SMS log await db.query( 'INSERT INTO sms_logs (to_phone, message, status) VALUES ($1,$2,$3)', [phone, message, 'sent'] ); return { success: true }; } catch (err) { await db.query( 'INSERT INTO sms_logs (to_phone, message, status) VALUES ($1,$2,$3)', [phone, message, 'failed'] ); return { success: false, error: err.message }; } } // Pre-built message templates const templates = { paymentReceived: (name, amount, unit, txnId) => `Dear ${name}, payment of KES ${amount} received for ${unit}. Ref: ${txnId}. Thank you!`, rentReminder: (name, unit, plot, paybill) => `Dear ${name}, your rent for ${unit} at ${plot} is due soon. Pay via M-Pesa Paybill ${paybill}.`, rentOverdue: (name, unit, months, balance, paybill) => `Dear ${name}, your rent for ${unit} is ${months} month(s) overdue. Balance: KES ${balance}. Pay via Paybill ${paybill}.`, }; module.exports = { sendSMS, templates };
3
Schedule Automatic Rent Reminders
Create backend/jobs/reminders.js. This runs daily at 9 AM and sends SMS reminders automatically:
backend/jobs/reminders.js
const cron = require('node-cron'); const db = require('../db'); const { sendSMS, templates } = require('../services/sms'); // Run every day at 9:00 AM cron.schedule('0 9 * * *', async () => { console.log('📅 Running daily rent reminders...'); const today = new Date().getDate(); // day of month (1-31) // 1. Tenants whose rent is due in 3 days const dueSoon = await db.query(` SELECT t.full_name, t.phone, h.unit_number, p.name as plot_name FROM tenants t JOIN houses h ON t.house_id = h.id JOIN plots p ON h.plot_id = p.id WHERE t.rent_due_day = $1 AND t.balance = 0 AND t.is_active = true `, [today + 3]); for (const tenant of dueSoon.rows) { await sendSMS( tenant.phone, templates.rentReminder(tenant.full_name, tenant.unit_number, tenant.plot_name, '400200') ); } // 2. Tenants with overdue rent (balance > 0) const overdue = await db.query(` SELECT t.full_name, t.phone, h.unit_number, h.rent_amount, t.balance FROM tenants t JOIN houses h ON t.house_id = h.id WHERE t.balance > 0 AND t.is_active = true `); for (const tenant of overdue.rows) { const months = Math.ceil(tenant.balance / tenant.rent_amount); await sendSMS( tenant.phone, templates.rentOverdue(tenant.full_name, tenant.unit_number, months, tenant.balance, '400200') ); } console.log(`✅ Sent reminders to ${dueSoon.rows.length + overdue.rows.length} tenants`); }); console.log('⏰ Reminder scheduler started');
Add require('./jobs/reminders') at the bottom of your server.js file to activate the scheduler when your server starts.
